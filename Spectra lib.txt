local function getESPColor()
    if _G.ESPData.selectedColor == "RGB" then
        local h = (tick() % 5) / 5
        return Color3.fromHSV(h,1,1)
    end
    local colors = {
        White = Color3.fromRGB(255,255,255),
        Black = Color3.fromRGB(0,0,0),
        Red = Color3.fromRGB(255,0,0),
        Green = Color3.fromRGB(0,255,0),
        Blue = Color3.fromRGB(0,170,255),
        Yellow = Color3.fromRGB(255,255,0),
        Pink = Color3.fromRGB(255,105,180),
        Purple = Color3.fromRGB(128,0,128)
    }
    return colors[_G.ESPData.selectedColor] or Color3.new(1,1,1)
end

_G.clearAllESP = function()
    for _,v in pairs(_G.ESPData.billboardGuis) do
        pcall(function() v:Destroy() end)
    end
    for _,v in pairs(_G.ESPData.highlights) do
        pcall(function() v:Destroy() end)
    end
    for _,v in pairs(_G.ESPData.lines) do
        pcall(function() v:Remove() end)
    end
    for _,v in pairs(_G.ESPData.connections) do
        pcall(function() v:Disconnect() end)
    end
    _G.ESPData.billboardGuis = {}
    _G.ESPData.highlights = {}
    _G.ESPData.lines = {}
    _G.ESPData.connections = {}
end

local function createNameESP(player)
    if player == Players.LocalPlayer then return end
    if not player.Character then return end
    if not player.Character:FindFirstChild("Head") then return end
    local gui = Instance.new("BillboardGui")
    gui.Size = UDim2.new(0,200,0,50)
    gui.StudsOffset = Vector3.new(0,3,0)
    gui.AlwaysOnTop = true
    gui.Parent = player.Character.Head
    local txt = Instance.new("TextLabel")
    txt.Size = UDim2.new(1,0,1,0)
    txt.BackgroundTransparency = 1
    txt.TextStrokeTransparency = 0.5
    txt.Font = Enum.Font.SourceSansBold
    txt.TextSize = 14
    txt.Text = player.Name.." | "..player.AccountAge.." days"
    txt.TextColor3 = getESPColor()
    txt.Parent = gui
    _G.ESPData.billboardGuis[player] = gui
end

local function createHighlightESP(player)
    if player == Players.LocalPlayer then return end
    if not player.Character then return end
    local h = Instance.new("Highlight")
    h.FillColor = getESPColor()
    h.FillTransparency = 0.5
    h.OutlineTransparency = 0
    h.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
    h.Parent = player.Character
    _G.ESPData.highlights[player] = h
end

local function createLineESP(player)
    if player == Players.LocalPlayer then return end
    local line = Drawing.new("Line")
    line.Thickness = 2
    line.Transparency = 1
    line.Visible = false
    _G.ESPData.lines[player] = line
end

_G.applyCurrentESP = function()
    _G.clearAllESP()
    if not _G.ESPData.espEnabled then return end
    for _,p in pairs(Players:GetPlayers()) do
        if _G.ESPData.espType == "Name + Age" then
            createNameESP(p)
        elseif _G.ESPData.espType == "Body (Highlight)" then
            createHighlightESP(p)
        elseif _G.ESPData.espType == "Lines" then
            createLineESP(p)
        end
    end
    table.insert(_G.ESPData.connections,
        RunService.RenderStepped:Connect(function()
            if not _G.ESPData.espEnabled then return end
            local color = getESPColor()
            if _G.ESPData.espType == "Lines" then
                for p,line in pairs(_G.ESPData.lines) do
                    if p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
                        local pos,onScreen =
                        Camera:WorldToViewportPoint(p.Character.HumanoidRootPart.Position)
                        line.Visible = onScreen
                        if onScreen then
                            line.From = Vector2.new(Camera.ViewportSize.X/2,Camera.ViewportSize.Y)
                            line.To = Vector2.new(pos.X,pos.Y)
                            line.Color = color
                        end
                    else
                        line.Visible = false
                    end
                end
            end
        end)
    )
end

    local function getPlayers()
        local t = {}

        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= Players.LocalPlayer then
                table.insert(t, p.Name)
            end
        end

        return t
    end

    local function getBodyBundleId(player)
        local ok, info = pcall(function()
            return player:GetCharacterAppearanceInfoAsync()
        end)

        if not ok or not info or not info.assets then
            return nil
        end

        for _, asset in ipairs(info.assets) do
            local at = asset.assetType
            local atName, atId

            if type(at) == "table" then
                atName = at.name
                atId = at.id
            else
                atName = tostring(at)
            end

            if atName == "Bundle" or atId == 8 then
                return tonumber(asset.id)
            end
        end

        return nil
    end

    dropdown = Tab3:AddDropdown({
        Name = "Choose Player",
        Options = getPlayers(),
        Callback = function(v)
            selectedPlayer = v
        end
    })

    local function refreshPlayerList()
        if not dropdown then return end
        local previous = selectedPlayer
        local options = getPlayers()
        local stillExists = false
        for _, name in ipairs(options) do
            if name == previous then stillExists = true break end
        end
        pcall(function() dropdown:Set(options) end)
        if stillExists then
            pcall(function() dropdown:Select(previous) end)
        else
            selectedPlayer = nil
        end
    end

    Tab3:AddButton({
        Name = "Refresh List",
        Callback = refreshPlayerList
    })

    Players.PlayerAdded:Connect(function()
        task.wait(0.5)
        refreshPlayerList()
    end)

    Players.PlayerRemoving:Connect(function()
        task.defer(refreshPlayerList)
    end)

        while task.wait(15) do
            refreshPlayerList()
        end
    end)

local function applyCarTurbo(vehicle)
    local seat = vehicle:FindFirstChild("Seats") and vehicle.Seats:FindFirstChild("VehicleSeat")
    if not seat then return end
    pcall(function()
        seat.Turbo = turboValue
    end)
    for _, v in pairs(seat:GetChildren()) do
        if v:IsA("NumberValue") and v.Name:lower():find("turbo") then
            v.Value = turboValue
        end
    end
end

local function applyBoatSpeed(vehicle)
    local body = vehicle:FindFirstChild("Body")
    if not body then return end
    local seat = body:FindFirstChild("VehicleSeat")
    if seat then
        pcall(function()
            seat.MaxSpeed = speedValue
        end)
    end
    for _, obj in pairs(body:GetDescendants()) do
        if obj:IsA("NumberValue") then
            obj.Value = speedValue
        end
    end
    if vehicle.PrimaryPart then
        vehicle.PrimaryPart.AssemblyLinearVelocity =
            vehicle.PrimaryPart.CFrame.LookVector * speedValue
    end
end

local function SkidFling(TargetPlayer, savedPos)
    pcall(function()
        local Character = plr.Character
        local Humanoid = Character and Character:FindFirstChildOfClass("Humanoid")
        local RootPart = Humanoid and Humanoid.RootPart
        local TCharacter = TargetPlayer.Character
        local THumanoid = TCharacter and TCharacter:FindFirstChildOfClass("Humanoid")
        local TRootPart = THumanoid and THumanoid.RootPart
        local THead = TCharacter and TCharacter:FindFirstChild("Head")

        if not (Character and Humanoid and RootPart and TCharacter and THumanoid and TRootPart) then return end

        if THead then
            Workspace.CurrentCamera.CameraSubject = THead
        elseif THumanoid then
            Workspace.CurrentCamera.CameraSubject = THumanoid
        end

        local FPos = function(BasePart, Pos, Ang)
            RootPart.CFrame = CFrame.new(BasePart.Position) * Pos * Ang
            Character:SetPrimaryPartCFrame(CFrame.new(BasePart.Position) * Pos * Ang)
            RootPart.Velocity = Vector3.new(3.15e9, 3.15e10, 3.15e9)
            RootPart.RotVelocity = Vector3.new(3.15e10, 3.15e10, 3.15e10)
        end

        local SFBasePart = function(BasePart)
            local Angle = 0
            repeat
                if not (RootPart and THumanoid) then break end
                if THumanoid.Sit then break end
                if not TargetPlayer.Character then break end
                if Humanoid.Health <= 0 then break end

                if BasePart.Velocity.Magnitude < 50 then
                    Angle = Angle + 100
                    FPos(BasePart, CFrame.new(0, 1.5, 0) + THumanoid.MoveDirection * BasePart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(Angle), 0, 0))
                    task.wait()
                    FPos(BasePart, CFrame.new(0, -1.5, 0) + THumanoid.MoveDirection * BasePart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(Angle), 0, 0))
                    task.wait()
                    FPos(BasePart, CFrame.new(2.25, 1.5, -2.25) + THumanoid.MoveDirection * BasePart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(Angle), 0, 0))
                    task.wait()
                    FPos(BasePart, CFrame.new(-2.25, -1.5, 2.25) + THumanoid.MoveDirection * BasePart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(Angle), 0, 0))
                    task.wait()
                    FPos(BasePart, CFrame.new(0, 1.5, 0) + THumanoid.MoveDirection, CFrame.Angles(math.rad(Angle), 0, 0))
                    task.wait()
                    FPos(BasePart, CFrame.new(0, -1.5, 0) + THumanoid.MoveDirection, CFrame.Angles(math.rad(Angle), 0, 0))
                    task.wait()
                else
                    FPos(BasePart, CFrame.new(0, 1.5, THumanoid.WalkSpeed), CFrame.Angles(math.rad(90), 0, 0))
                    task.wait()
                    FPos(BasePart, CFrame.new(0, -1.5, -THumanoid.WalkSpeed), CFrame.Angles(0, 0, 0))
                    task.wait()
                    FPos(BasePart, CFrame.new(0, 1.5, THumanoid.WalkSpeed), CFrame.Angles(math.rad(90), 0, 0))
                    task.wait()
                    FPos(BasePart, CFrame.new(0, 1.5, TRootPart.Velocity.Magnitude / 1.25), CFrame.Angles(math.rad(90), 0, 0))
                    task.wait()
                    FPos(BasePart, CFrame.new(0, -1.5, -TRootPart.Velocity.Magnitude / 1.25), CFrame.Angles(0, 0, 0))
                    task.wait()
                    FPos(BasePart, CFrame.new(0, 1.5, TRootPart.Velocity.Magnitude / 1.25), CFrame.Angles(math.rad(90), 0, 0))
                    task.wait()
                    FPos(BasePart, CFrame.new(0, -1.5, 0), CFrame.Angles(math.rad(90), 0, 0))
                    task.wait()
                    FPos(BasePart, CFrame.new(0, -1.5, 0), CFrame.Angles(0, 0, 0))
                    task.wait()
                    FPos(BasePart, CFrame.new(0, -1.5, 0), CFrame.Angles(math.rad(-90), 0, 0))
                    task.wait()
                    FPos(BasePart, CFrame.new(0, -1.5, 0), CFrame.Angles(0, 0, 0))
                    task.wait()
                end
            until BasePart.Velocity.Magnitude > 500 or not BasePart.Parent or THumanoid.Sit or Humanoid.Health <= 0
        end

        local BV = Instance.new("BodyVelocity")
        BV.Parent = RootPart
        BV.Velocity = Vector3.new(9e8, 9e8, 9e8)
        BV.MaxForce = Vector3.new(1/0, 1/0, 1/0)

        Humanoid:SetStateEnabled(Enum.HumanoidStateType.Seated, false)

        if TRootPart and THead then
            if (TRootPart.CFrame.p - THead.CFrame.p).Magnitude > 5 then
                SFBasePart(THead)
            else
                SFBasePart(TRootPart)
            end
        elseif TRootPart then
            SFBasePart(TRootPart)
        elseif THead then
            SFBasePart(THead)

local function doFlingSofa()

    local tRoot = target.Character:FindFirstChild("HumanoidRootPart")
    local tHum = target.Character:FindFirstChildOfClass("Humanoid")

    if not (root and hum and tRoot and tHum) then return end

    local clearRemote = ReplicatedStorage.RE:FindFirstChild("1Clea1rTool1s")
    if clearRemote then clearRemote:FireServer("ClearAllTools") end

    local tool = LocalPlayer.Backpack:FindFirstChildOfClass("Tool")

    workspace.FallenPartsDestroyHeight = -math.huge

    local bv = Instance.new("BodyVelocity")
    bv.Name = "FlingForce"
    bv.Velocity = Vector3.new(9e8,9e8,9e8)
    bv.MaxForce = Vector3.new(math.huge,math.huge,math.huge)
    bv.Parent = root

    hum:SetStateEnabled(Enum.HumanoidStateType.Seated,false)
    hum.PlatformStand = false

    cam.CameraSubject = tRoot

    local angle = 0
    local t = tick()

    while tick() - t < 3 and target and target.Character and target.Character:FindFirstChildOfClass("Humanoid") do
        tRoot = target.Character:FindFirstChild("HumanoidRootPart")
        if not tRoot then break end
        angle += 30
        root.CFrame = CFrame.new(tRoot.Position + Vector3.new(0,1,0)) * CFrame.Angles(math.rad(angle),0,0)
        root.Velocity = Vector3.new(9e8,9e8,9e8)
        root.RotVelocity = Vector3.new(9e8,9e8,9e8)
        task.wait()
    end

    bv:Destroy()
    hum:SetStateEnabled(Enum.HumanoidStateType.Seated,true)
    hum.PlatformStand = false
    root.CFrame = original
    cam.CameraSubject = hum

    for _,p in pairs(char:GetDescendants()) do
        if p:IsA("BasePart") then
            p.Velocity = Vector3.zero
            p.RotVelocity = Vector3.zero
        end
    end

    hum:UnequipTools()
    if toolRemote then toolRemote:InvokeServer("PickingTools","Couch") end
end

local LOOP_ENABLED = false
local raio = 10
local passoAngulo = math.rad(30)
local abertura = math.rad(70)
local anguloAtual = 0

local function ExecuteProp()
    if not character or not HRP then return end

    pcall(function()
        game.ReplicatedStorage.RE["1Too1l"]:InvokeServer("PickingTools", "PropMaker")
    end)

    local tool = plr.Backpack:FindFirstChild("PropMaker")
    if not tool then return end
    tool.Parent = character

    local frente = HRP.CFrame.LookVector
    local anguloFrente = math.atan2(frente.Z, frente.X)

    local anguloValido = false
    local tentativas = 0
    local anguloTeste = anguloAtual
    while not anguloValido and tentativas < 100 do
        local diff = (anguloTeste - anguloFrente + math.pi) % (2 * math.pi) - math.pi
        if math.abs(diff) < abertura then
            anguloTeste = (anguloTeste + passoAngulo) % (2 * math.pi)
            tentativas = tentativas + 1
        else
            anguloValido = true
            anguloAtual = anguloTeste
        end
    end

    local x = math.cos(anguloAtual) * raio
    local z = math.sin(anguloAtual) * raio
    local pos = HRP.Position + Vector3.new(x, 0, z)
    local posFinal = Vector3.new(pos.X, HRP.Position.Y, pos.Z)

    local args = {
        workspace:WaitForChild("Model"):WaitForChild("Sidewalk"),
        posFinal
    }

    pcall(function()
        tool:WaitForChild("Tool_PropMake"):FireServer(unpack(args))
    end)

    anguloAtual = (anguloAtual + passoAngulo) % (2 * math.pi)
end

local function StartPropLoop()
    task.spawn(function()
        while LOOP_ENABLED do
            pcall(ExecuteProp)
            task.wait(0.2)
        end
    end)
end

                    pcall(function() prop.ChangePropColor:InvokeServer(cor) end)

            end
        end)

        workspace:FindFirstChild(
            "WorkspaceCom"
        )

    if not WorkspaceCom then
        return Result
    end

    local PlayerName =
        string.lower(
            Player.Name
        )

    for _, Object in ipairs(
        WorkspaceCom:GetDescendants()
    ) do

        local Remote =
            Object:FindFirstChild(
                "SetCurrentCFrame"
            )

        if Remote
            and Remote:IsA(
                "RemoteFunction"
            ) then

            local ObjectName =
                string.lower(
                    Object.Name
                )

            if ObjectName:find(
                PlayerName,
                1,
                true
            ) then

                table.insert(
                    Result,
                    Object
                )

            end
        end
    end

    table.sort(
        Result,
        function(a, b)

            return a.Name < b.Name

        end
    )

    return Result
end

local function GetUprightRotation()

    if not PropSuitHRP then
        return CFrame.new()
    end

    local Look =
        PropSuitHRP.CFrame.LookVector

    local FlatLook =
        Vector3.new(
            Look.X,
            0,
            Look.Z
        )

    if FlatLook.Magnitude < 0.001 then

        FlatLook =
            Vector3.new(
                0,
                0,
                -1
            )

    else

        FlatLook =
            FlatLook.Unit

    end

    local Yaw =
        CFrame.lookAt(
            Vector3.zero,
            FlatLook,
            Vector3.yAxis
        )

    local PropRotation =
        CFrame.Angles(

            math.rad(
                PropSuitConfig.RotationX
            ),

            math.rad(
                PropSuitConfig.RotationY
            ),

            math.rad(
                PropSuitConfig.RotationZ
            )
        )

    return Yaw * PropRotation
end

local function GetPropSuitTargetCFrame(
    BodyPart,
    Index
)

    if not PropSuitHRP
        or not BodyPart then

        return nil
    end

    local Relative =
        PropSuitHRP.CFrame:ToObjectSpace(
            BodyPart.CFrame
        )

    local Position =
        Relative.Position

    Position = Vector3.new(

        Position.X *
            PropSuitConfig.BodySpacing,

        Position.Y *
            PropSuitConfig.BodySpacing,

        Position.Z *
            PropSuitConfig.BodySpacing
    )

    local IsR15 =
        PropSuitChar
        and PropSuitChar:FindFirstChild(
            "UpperTorso"
        ) ~= nil

    if not IsR15 then

        local Extra =
            R6ExtraOffsets[Index]

        if Extra then

            Position +=
                Extra *
                PropSuitConfig.BodySpacing

        end
    end

    if Index == 1 then

        Position += Vector3.new(
            0,
            PropSuitConfig.HeadGap,
            0
        )

    end

    Position += Vector3.new(
        0,
        PropSuitConfig.HeightOffset,
        0
    )

    local FinalPosition =
        PropSuitHRP.CFrame:PointToWorldSpace(
            Position
        )

    FinalPosition =
        FinalPosition
        - PropSuitHRP.CFrame.LookVector
        * PropSuitConfig.BehindDistance

    local FinalRotation =
        GetUprightRotation()

    return CFrame.new(
        FinalPosition
    ) * FinalRotation
end

local function MovePropSuit(
    Prop,
    BodyPart,
    Index
)

    if not Prop
        or not BodyPart then

        return
    end

    local Remote =
        GetPropSuitRemote(Prop)

    if not Remote then
        return
    end

    local Target =
        GetPropSuitTargetCFrame(
            BodyPart,
            Index
        )

    if not Target then
        return
    end

    pcall(function()

        Remote:InvokeServer(
            Target
        )

    end)
end

local function UpdatePropSuit()

    if not PropSuitActive then
        return
    end

    if not PropSuitChar
        or not PropSuitHumanoid
        or not PropSuitHRP then

        return
    end

    if PropSuitHumanoid.Health <= 0 then
        return
    end

    local Now =
        os.clock()

    if Now - PropSuitLastScan >=
        PropSuitConfig.RescanTime then

        local NewProps =
            GetPropSuitProps()

        if #NewProps > 0 then

            PropSuitProps =
                NewProps

        end

        PropSuitLastScan =
            Now
    end

    if #PropSuitProps == 0 then
        return
    end

    local BodyParts =
        GetPropSuitBodyParts()

    if not BodyParts then
        return
    end

    local Amount =
        math.min(
            #PropSuitProps,
            #BodyParts
        )

    for Index = 1, Amount do

        local Prop =
            PropSuitProps[Index]

        local BodyPart =
            BodyParts[Index]

        if Prop and BodyPart then

            MovePropSuit(
                Prop,
                BodyPart,
                Index
            )

        end
    end
end

function StartPropSuit()

    if PropSuitConnection then

        PropSuitConnection:Disconnect()
        PropSuitConnection = nil

    end

    if not PropSuitActive then
        return
    end

    PropSuitProps =
        GetPropSuitProps()

    PropSuitLastScan =
        os.clock()

    PropSuitTimer = 0

    PropSuitConnection =
        RunService.Heartbeat:Connect(
            function(dt)

                if not PropSuitActive then
                    return
                end

                PropSuitTimer += dt

                local Interval =
                    1 /
                    math.max(
                        PropSuitConfig.UpdateRate,
                        1
                    )

                if PropSuitTimer < Interval then
                    return
                end

                PropSuitTimer = 0

                UpdatePropSuit()

            end
        )
end

function StopPropSuit()

    PropSuitActive = false

    if PropSuitConnection then

        PropSuitConnection:Disconnect()
        PropSuitConnection = nil

    end

    PropSuitProps = {}
end

local function lim(v, minValue, maxValue)
    v = tonumber(v)
    if v == nil then
        return minValue
    end
    return math.clamp(v, minValue, maxValue)
end

local function normId(value)
    local s = tostring(value or "")
    s = s:gsub("^%s+", ""):gsub("%s+$", "")
    if s == "" then
        return ""
    end

    local numeric = s:match("^rbxassetid://(%d+)$")
    if numeric then
        return numeric
    end

    local onlyNumber = s:match("^(%d+)$")
    if onlyNumber then
        return onlyNumber
    end

    return s
end

local function nomeDe(idPorNome, nomes, valor, fallback)
    local alvo = normId(valor)

    if alvo ~= "" then
        for _, nome in ipairs(nomes or {}) do
            if normId(idPorNome[nome]) == alvo then
                return nome
            end
        end
    end

    return fallback
end

local function adicionarImagem(nomes, idPorNome, texto, prefixo)
    local src = (tostring(texto or ""):gsub("^%s+", ""):gsub("%s+$", ""))
    if src == "" then return nil end
    local id
    if src:match("^https?://") or src:match("^rbxassetid://") then
        id = src
    elseif src:match("^%d+$") then
        id = "rbxassetid://" .. src
    else
        return nil
    end
    local alvo = normId(id)
    for _, nome in ipairs(nomes) do
        if normId(idPorNome[nome]) == alvo then return nome end
    end
    local n, nome = #nomes + 1, nil
    repeat nome = prefixo .. " " .. n; n = n + 1 until idPorNome[nome] == nil
    table.insert(nomes, nome)
    idPorNome[nome] = id
    return nome
end

feTab:AddButton({
    Name = "Stop FEs",
    Callback = function()

        local Players = game:GetService("Players")
        local ReplicatedStorage = game:GetService("ReplicatedStorage")
        local SoundService = game:GetService("SoundService")
        local player = Players.LocalPlayer

        local character = player.Character
        local humanoid = character
            and character:FindFirstChildOfClass("Humanoid")
        local root = character
            and character:FindFirstChild("HumanoidRootPart")

        local stopped = 0

        local function safeDestroy(obj)
            if obj and obj.Parent then
                pcall(function()
                    obj:Destroy()
                end)
            end
        end

        pcall(function()
            local env = getgenv and getgenv() or _G
            local cleanups = env.__SSJ_Cleanups

            if type(cleanups) == "table" then
                for _, cleanup in ipairs(cleanups) do
                    if type(cleanup) == "function" then
                        pcall(cleanup)
                    end
                end

                env.__SSJ_Cleanups = {}
                stopped += 1
            end
        end)

        if character then
            for _, obj in ipairs(character:GetDescendants()) do
                if obj:IsA("ParticleEmitter") then
                    if obj:GetAttribute("GokuSSJ2")
                    or obj:GetAttribute("GokuBodyKi")
                    or obj:GetAttribute("GokuCharge") then
                        safeDestroy(obj)
                    end
                elseif obj.Name == "AuraAttachment"
                or obj.Name == "BodyKiAtt"
                or obj.Name == "ChargeOut" then
                    safeDestroy(obj)
                end
            end
        end

        for _, obj in ipairs(workspace:GetDescendants()) do
            if obj.Name == "GokuSSJ2_Sphere"
            or obj.Name == "GokuSSJ2_Inner" then
                safeDestroy(obj)
            end
        end

        pcall(function()
            local coreGui = game:GetService("CoreGui")
            safeDestroy(coreGui:FindFirstChild("SonicGUI"))
        end)

        pcall(function()
            local playerGui = player:FindFirstChild("PlayerGui")
            if playerGui then
                safeDestroy(playerGui:FindFirstChild("SonicGUI"))
            end
        end)

        if character then
            for _, obj in ipairs(character:GetDescendants()) do
                if obj.Name == "SonicLightning"
                or obj.Name == "SonicRumble"
                or obj.Name == "SonicZap"
                or obj.Name == "PortalRingHand" then
                    safeDestroy(obj)
                end
            end
        end

        pcall(function()
            local backpack = player:FindFirstChild("Backpack")
            if backpack then
                safeDestroy(backpack:FindFirstChild("Portal Ring"))
            end
        end)

        for _, obj in ipairs(workspace:GetDescendants()) do
            if obj.Name == "PortalVisual"
            or obj.Name == "ThrownPortalRing" then
                safeDestroy(obj)
            end
        end

        pcall(function()
            safeDestroy(
                game:GetService("CoreGui"):FindFirstChild("XRayESPGui")
            )
        end)

        safeDestroy(
            ReplicatedStorage:FindFirstChild("BZn2q91BzN")
        )

        if player:FindFirstChild("PlayerGui") then
            safeDestroy(
                player.PlayerGui:FindFirstChild("BotoesRedondos")
            )
        end

        for _, obj in ipairs(workspace:GetDescendants()) do
            if obj:IsA("ParticleEmitter")
            and obj.Texture == "rbxassetid://258128463" then
                local parent = obj.Parent

                if parent and parent:IsA("Part")
                and parent.Anchored
                and parent.Transparency == 1 then
                    safeDestroy(parent)
                end
            end
        end

        safeDestroy(workspace:FindFirstChild("SunModel"))

        for _, obj in ipairs(SoundService:GetChildren()) do
            if obj:IsA("Sound") then
                if obj.Name == "SpectraMusic"
                or obj.Name == "GokuChargeAudio"
                or obj.Name == "AnimeRevengeAudio" then
                    pcall(function()
                        obj:Stop()
                        obj:Destroy()
                    end)
                end
            end
        end

        if humanoid then
            humanoid.PlatformStand = false

            if humanoid.WalkSpeed > 16 then
                humanoid.WalkSpeed = 16
            end

            local animator = humanoid:FindFirstChildOfClass("Animator")
            if animator then
                for _, track in ipairs(
                    animator:GetPlayingAnimationTracks()
                ) do
                    pcall(function()
                        track:Stop()
                    end)
                end
            end
        end

        if root then
            for _, obj in ipairs(root:GetChildren()) do
                if obj.Name == "VelocityHandler"
                or obj.Name == "GyroHandler" then
                    safeDestroy(obj)
                end
            end
        end

        local camera = workspace.CurrentCamera

        if camera then
            camera.CameraType = Enum.CameraType.Custom

            if humanoid then
                camera.CameraSubject = humanoid
            end

            camera.FieldOfView = 70
        end

    end
})

local function PlayEmote(id)
pcall(function()
if CurrentTrack then
CurrentTrack:Stop(0.1)
end

	local animId

	local ok, result = pcall(function()
		return game:GetObjects("rbxassetid://" .. tostring(id))
	end)

	if ok and result and result[1] and result[1]:IsA("Animation") then
		animId = result[1].AnimationId
	else
		animId = "rbxassetid://" .. tostring(id)
	end

	local anim = Instance.new("Animation")
	anim.AnimationId = animId

	local track = humanoid:LoadAnimation(anim)
	track.Priority = Enum.AnimationPriority.Action4
	track.Looped = LoopEmote
	track:Play(0.1)
	track:AdjustSpeed(GetSpeed())

	CurrentTrack = track
	CurrentEmoteId = id

	track.Stopped:Connect(function()
		if CurrentTrack == track then
			CurrentTrack = nil
			CurrentEmoteId = nil
		end
	end)
end)

end

end)

	EmotePager:SetLoading(
		false,
		"Failed to download emotes."
	)

	warn("[Spectra] Failed to download emote list.")
	return
end

EmotePager:SetLoading(true, "Processing emote list...")

local decodeSuccess, decoded = pcall(function()
	return game:GetService("HttpService"):JSONDecode(response)
end)

if not decodeSuccess then
	EmotePager:SetLoading(
		false,
		"Failed to decode emotes.json."
	)

	warn("[Spectra] Failed to decode emotes.json.")
	return
end

local Emotes

if type(decoded) == "table"
	and type(decoded.data) == "table" then

	Emotes = decoded.data

elseif type(decoded) == "table"
	and type(decoded.emotes) == "table" then

	Emotes = decoded.emotes

elseif type(decoded) == "table"
	and #decoded > 0 then

	Emotes = decoded
end

if type(Emotes) ~= "table" then
	EmotePager:SetLoading(
		false,
		"Invalid JSON format."
	)

	warn("[Spectra] Invalid JSON format.")
	return
end

EmotePager:SetItems(Emotes)

end

                end

                return OldNamecall(self, ...)
            end))

            pcall(function()
                if type(getgenv().AntiHTTPSpy_TrustCurrentHooks) == "function" then
                    getgenv().AntiHTTPSpy_TrustCurrentHooks()
                end
                if type(getgenv().AntiHTTPSpy_WhitelistFunction) == "function" then
                    getgenv().AntiHTTPSpy_WhitelistFunction(OldNamecall)
                end
            end)

            ToolHookConnection = {
                Disconnect = function()
                    isHooked = false
                    if OldNamecall then
                        pcall(function()
                            hookmetamethod(game, "__namecall", OldNamecall)
                        end)

                        pcall(function()
                            if type(getgenv().AntiHTTPSpy_TrustCurrentHooks) == "function" then
                                getgenv().AntiHTTPSpy_TrustCurrentHooks()
                            end
                        end)

                        OldNamecall = nil
                    end
                end
            }

            local function unlockViaRemote(gpId, gpName)
                local remotesToTry = {
                    "PromptGamepassPurchase",
                    "ShowGamepass",
                    "GamepassPurchased",
                    "IM_GamepassAll",
                    "IM_GamepassClear",
                }

                for _, remoteName in ipairs(remotesToTry) do
                    local success = pcall(function()
                        local remote = Remotes.get(remoteName)
                        if remote then
                            remote:fireServer(gpId, player.UserId, "ForceUnlock")
                            return true
                        end
                        return false
                    end)

                    if success then
                        return true
                    end
                end

                return false
            end

            local function unlockViaUnlockable(gpName)
                local success = pcall(function()
                    local featureName = ("Feature_%*"):format(gpName)
                    UnlockableController.UnlockFeature(featureName)
                    return true
                end)
                return success
            end

            local function forceDataUnlock(gpId, gpName)
                local success, replica = pcall(GlobalReplicatedDataController.WaitForReplica)
                if not success or not replica then
                    return false
                end

                local userId = tostring(player.UserId)
                local userData = replica.Data[userId]

                if not userData then
                    return false
                end

                local modified = false

                pcall(function()
                    if userData.profile then
                        if userData.profile.gamepasses then
                            userData.profile.gamepasses[tostring(gpId)] = true
                            modified = true
                        end
                        if userData.profile.gamepassLegacyOwnership then
                            userData.profile.gamepassLegacyOwnership[tostring(gpId)] = true
                            modified = true
                        end
                        if userData.profile.devProducts then
                            userData.profile.devProducts[tostring(gpId)] = true
                            modified = true
                        end
                    end

                    if userData.session then
                        if userData.session.gamepasses then
                            userData.session.gamepasses[tostring(gpId)] = true
                            modified = true
                        end
                        if userData.session.devProducts then
                            userData.session.devProducts[tostring(gpId)] = true
                            modified = true
                        end
                    end
                end)

                return modified
            end

            local function forceAttributes(gpName, gpId)
                local success = pcall(function()
                    player:SetAttribute("Gamepass_" .. gpName, true)
                    player:SetAttribute("Has_" .. gpName, true)
                    player:SetAttribute("Unlocked_" .. gpName, true)
                    player:SetAttribute(gpName, true)
                    player:SetAttribute("Pass_" .. gpName, true)

                    local dataFolder = player:FindFirstChild("Data")
                    if dataFolder then
                        for _, child in ipairs(dataFolder:GetChildren()) do
                            if child:IsA("BoolValue") and string.find(child.Name, gpName) then
                                child.Value = true
                            end
                            if child:IsA("NumberValue") and string.find(child.Name, gpName) then
                                child.Value = 1
                            end
                        end
                    end

                    return true
                end)
                return success
            end

            local function forceUIUpdate()
                pcall(function()
                    local allGamepasses = ReplicatedStorage:FindFirstChild("Modules"):FindFirstChild("Client"):FindFirstChild("Components"):FindFirstChild("UI"):FindFirstChild("Shop"):FindFirstChild("AllGamepasses")
                    if allGamepasses then
                        local comp = require(allGamepasses)
                        if comp and comp.Open then
                            comp.isGenerated = false
                            comp.Open()
                        end
                    end
                end)

                pcall(function()
                    for _, obj in ipairs(game:GetService("CollectionService"):GetTagged("ShowGamepass")) do
                        if obj:IsA("GuiObject") then
                            obj.Visible = true
                            pcall(function()
                                local comp = require(obj.Parent:FindFirstChild("Component"))
                                if comp and comp.Start then
                                    comp.Start()
                                end
                            end)
                        end
                    end
                end)

                pcall(function()
                    local vehicles = workspace:FindFirstChild("Vehicles")
                    if vehicles then
                        for _, vehicle in ipairs(vehicles:GetChildren()) do
                            pcall(function()
                                vehicle.Visible = true
                                for _, part in ipairs(vehicle:GetDescendants()) do
                                    if part:IsA("BasePart") then
                                        part.Transparency = 0
                                        part.Visible = true
                                    end
                                end
                            end)
                        end
                    end
                end)
            end

            if not _G.EmmitersScriptLoaded then
                _G.EmmitersScriptLoaded = true

                task.spawn(function()
                    local emmiterListPath = player:WaitForChild("PlayerGui")
                        :WaitForChild("NoResetGUIHandler")
                        :WaitForChild("AvatarEditorHolder")
                        :WaitForChild("AvatarEditorMenu")
                        :WaitForChild("Catalog")
                        :WaitForChild("Container")
                        :WaitForChild("EmmiterList")

                    local ApplyEmmiter = ReplicatedStorage:WaitForChild("Remotes"):WaitForChild("ApplyEmmiter")
                    local currentActiveIndicator = nil
                    local activeEffectCode = nil

                    local function cleanOldButtons()
                        for _, item in ipairs(emmiterListPath:GetChildren()) do
                            if item:IsA("GuiObject") then
                                local oldButton = item:FindFirstChild("EquipButtonOverlay")
                                if oldButton then oldButton:Destroy() end
                            end
                        end
                    end

                    local function setupEquipButtons()
                        cleanOldButtons()
                        for _, item in ipairs(emmiterListPath:GetChildren()) do
                            if item:IsA("GuiObject") and item.Name ~= "UIGridLayout" then
                                local idPart, codePart = string.match(item.Name, "^(%d+)_(.+)$")
                                if idPart and codePart then
                                    local cleanId = tostring(tonumber(idPart))

                                    local equipButton = Instance.new("TextButton")
                                    equipButton.Name = "EquipButtonOverlay"
                                    equipButton.Size = UDim2.new(1, 0, 1, 0)
                                    equipButton.BackgroundTransparency = 1
                                    equipButton.Text = ""
                                    equipButton.ZIndex = item.ZIndex + 5
                                    equipButton.Parent = item

                                    local indicator = Instance.new("ImageLabel")
                                    indicator.Name = "EquipIndicator"
                                    indicator.Size = UDim2.new(0.6, 0, 0.6, 0)
                                    indicator.Position = UDim2.new(0.2, 0, 0.2, 0)
                                    indicator.BackgroundTransparency = 1
                                    indicator.Image = "rbxassetid://6935798751"
                                    indicator.ImageColor3 = Color3.fromRGB(0, 255, 100)
                                    indicator.Visible = false
                                    indicator.ZIndex = equipButton.ZIndex + 1
                                    indicator.Parent = equipButton

                                    if activeEffectCode == codePart then
                                        indicator.Visible = true
                                        currentActiveIndicator = indicator
                                    end

                                    equipButton.MouseButton1Click:Connect(function()
                                        if activeEffectCode == codePart then
                                            pcall(function()
                                                ApplyEmmiter:InvokeServer(0, "Clear")
                                            end)
                                            indicator.Visible = false
                                            currentActiveIndicator = nil
                                            activeEffectCode = nil
                                        else
                                            local args = { [1] = cleanId, [2] = codePart }
                                            local success = pcall(function()
                                                ApplyEmmiter:InvokeServer(unpack(args))
                                            end)
                                            if success then
                                                if currentActiveIndicator then
                                                    currentActiveIndicator.Visible = false
                                                end
                                                indicator.Visible = true
                                                currentActiveIndicator = indicator
                                                activeEffectCode = codePart
                                            end
                                        end
                                    end)
                                end
                            end
                        end
                    end

                    setupEquipButtons()
                    emmiterListPath.ChildAdded:Connect(function()
                        task.wait(0.1)
                        setupEquipButtons()
                    end)
                end)
            end

            local allGamepasses = {}
            for name, data in pairs(Gamepasses.All) do
                local id = Gamepasses.GetId(data)
                table.insert(allGamepasses, {name = name, id = id})
            end

            table.insert(allGamepasses, {name = "ChangeCarColor",      id = 1459820822})
            table.insert(allGamepasses, {name = "CarColour",           id = 1459820822})
            table.insert(allGamepasses, {name = "HouseMenu",           id = 3713054065})
            table.insert(allGamepasses, {name = "HouseMenuAlt",        id = 3557854012})
            table.insert(allGamepasses, {name = "Lot",                 id = 82773869})
            table.insert(allGamepasses, {name = "Firework_HeartBurst", id = 3606424793})
            table.insert(allGamepasses, {name = "StickyBomb",          id = 3606700071})

            table.sort(allGamepasses, function(a, b) return a.name < b.name end)

            for i, gp in ipairs(allGamepasses) do
                if not UnlockGamepassRunning then
                    return
                end

                local unlocked = false

                if not unlocked then
                    unlocked = unlockViaRemote(gp.id, gp.name)
                    task.wait(0.3)
                end

                if not UnlockGamepassRunning then return end

                if not unlocked then
                    unlocked = unlockViaUnlockable(gp.name)
                    task.wait(0.2)
                end

                if not UnlockGamepassRunning then return end

                if not unlocked then
                    unlocked = forceDataUnlock(gp.id, gp.name)
                    task.wait(0.1)
                end

                if not UnlockGamepassRunning then return end

                if not unlocked then
                    unlocked = forceAttributes(gp.name, gp.id)
                    task.wait(0.1)
                end

                task.wait(0.2)
            end

            if not UnlockGamepassRunning then return end
            forceUIUpdate()
        end)
    end
